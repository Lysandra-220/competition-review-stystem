#!/bin/bash
# 面向学院竞赛报名评审系统 Ubuntu 一键部署脚本
echo "========== 开始部署竞赛管理系统 =========="
# 更新软件源
sudo apt update
# 安装 Python 环境
sudo apt install -y python3 python3-pip
# 安装项目依赖
pip3 install flask pandas openpyxl werkzeug gunicorn
# 赋予权限
chmod -R 755 .
# 后台启动项目
gunicorn -w 3 -b 0.0.0.0:5000 app:app -D
echo "========== 部署完成 =========="
echo "访问地址：服务器IP:5000"
echo "停止服务命令：pkill -f gunicorn"
